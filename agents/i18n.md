# Internationalization (i18n) Specification

## Overview

Internationalization support for Spanish (es) and English (en). Next.js uses next-intl for routing (e.g., `/es/...`, `/en/...`). Mobile app uses a simple key-based translation system.

---

## Key Structure

### Categories

```
i18n/
├── common.json       # Shared keys (buttons, labels, errors)
├── auth.json         # Authentication related
├── parkings.json     # Parkings module
├── reservations.json # Reservations module
├── inspections.json  # Inspections module
└── errors.json       # Error messages
```

## Mobile App (React Native)

```typescript
// i18n/index.ts
import es from './locales/es';
import en from './locales/en';

const translations = { es, en };

export const t = (key: string, locale: string = 'es'): string => {
  const keys = key.split('.');
  let value = translations[locale];
  
  for (const k of keys) {
    value = value?.[k];
  }
  
  return value || key;
};

export const setLocale = (locale: string) => {
  // Update context/state
};
```

## Usage Examples

### Next.js (Server Components)

```typescript
import { getTranslations } from 'next-intl/server';

export default async function Page() {
  const t = await getTranslations('parkings');
  
  return (
    <h1>{t('lots.title')}</h1>
  );
}
```

### Next.js (Client Components)

```typescript
'use client';
import { useTranslations } from 'next-intl/react';

export function ParkingForm() {
  const t = useTranslations('parkings');
  
  return (
    <Button>{t('vehicles.add')}</Button>
  );
}
```

### React Native

```typescript
import { t } from '../i18n';

export function VehicleCard({ vehicle }) {
  return (
    <View>
      <Text>{vehicle.plate}</Text>
      <Text>{t('parkings.status.available')}</Text>
    </View>
  );
}
```
